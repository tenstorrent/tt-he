# ---- repo config: one source of truth for where tt-metalium lives (see config.defaults.env) ----
# Precedence matches env.sh: environment/cache > config.local.env > config.defaults.env > autodetect.
function(he_config_value key out_var)
    foreach(cfg "${CMAKE_CURRENT_SOURCE_DIR}/config.local.env"
                "${CMAKE_CURRENT_SOURCE_DIR}/config.defaults.env")
        if(EXISTS "${cfg}")
            file(STRINGS "${cfg}" _hit REGEX "^[ \t]*${key}=")
            if(_hit)
                list(GET _hit 0 _line)
                string(REGEX REPLACE "^[ \t]*${key}=" "" _val "${_line}")
                string(STRIP "${_val}" _val)
                if(NOT _val STREQUAL "")
                    set(${out_var} "${_val}" PARENT_SCOPE)
                    return()
                endif()
            endif()
        endif()
    endforeach()
    set(${out_var} "" PARENT_SCOPE)
endfunction()

function(he_prefix_usable prefix out_var)
    if(prefix STREQUAL "")
        set(${out_var} FALSE PARENT_SCOPE)
        return()
    endif()
    if(EXISTS "${prefix}/lib/cmake/tt-metalium/tt-metalium-config.cmake"
       AND IS_DIRECTORY "${prefix}/libexec/tt-metalium")
        set(${out_var} TRUE PARENT_SCOPE)
    else()
        set(${out_var} FALSE PARENT_SCOPE)
    endif()
endfunction()

    he_prefix_usable("${HE_TTM_PREFIX}" _ok)
    if(NOT _ok)
        # Fail here, naming the cause -- otherwise this surfaces as an unreadable find_package error.
        set(_hint "")
        if(EXISTS "${HE_TTM_PREFIX}/CMakeCache.txt")
            string(CONCAT _hint
                "\n  That is a tt-metal BUILD TREE, not an install: Metalium.cmake is"
                "\n  generated by `cmake --install`, so find_package can never use it.")
        endif()
        message(FATAL_ERROR
            "Configured tt-metalium prefix is not a usable install:\n  ${HE_TTM_PREFIX}${_hint}\n"
            "A prefix must contain BOTH:\n"
            "  host/lib/cmake/tt-metalium/tt-metalium-config.cmake\n"
            "  libexec/tt-metalium/\n"
            "Fix it with:  sh ${CMAKE_CURRENT_SOURCE_DIR}/tools/setup.sh\n"
            "Diagnose:     sh ${CMAKE_CURRENT_SOURCE_DIR}/tools/doctor.sh")
    endif()
    list(PREPEND CMAKE_PREFIX_PATH "${HE_TTM_PREFIX}")
    message(STATUS "tt-metalium prefix (from repo config): ${HE_TTM_PREFIX}")
endif()
endif()  # NOT HE_HOST_ONLY
